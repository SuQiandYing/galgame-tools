     * 用 U+2009 细空格，并写成   转义形式：普通空格在正文里大量出现，
     * 用作哨兵会把几乎所有带空格的 title 误判成已处理而跳过翻译；写转义还能
     * 避免编辑时字面细空格被当成普通空格丢掉。
     * 声明放在最前：受保护列() 要用它从 title 取表头原文，那段代码位置更靠上，

    const 标记 = ' ';

     * 1. 结构保护层
     *    ErogameScape 的用户投稿内容是裸文本节点，直接躺在容器里，没有包裹标签。

    /* 完全不进入的标签 */
    const 禁入标签 = new Set(['SCRIPT', 'STYLE', 'NOSCRIPT', 'TEXTAREA', 'CODE', 'PRE', 'SVG', 'CANVAS', 'IFRAME', 'OBJECT']);

    /* 纯投稿容器：整棵子树都是用户写的字，完全不进入。 */
    const 完全禁入 = [
        '#memo',            // 长评正文
        '#memo_main',

    /* 混合容器：真实结构是「用户正文 = 裸文本节点，界面文字 = 包在子元素里」。
     *   <div class="even comment_pagging">
     *     <span class="red bold">10</span>点            ← 界面（裸 "点"）
     *     <span class="bold"><a>作品名</a></span>        ← 专有名词
     *     想いは渡月橋をこえられない →                    ← 用户短评（裸文本）
     *     <a href="memo.php...">長文感想(1848)</a>       ← 界面
     *     <div class="play_time">総プレイ時間 : 4h</div>  ← 界面
     *   </div>
     * 所以不能整块拦掉，否则界面文字全留日文。
     */
    const 混合容器 = [
        '.comment_pagging',
        '.line-height_comment',

    /* 混合容器里唯一允许翻译的裸文本片段（真实页面确认：得分后面那个「点」）。
     * 其余裸文本一律视为用户投稿，保持原文。纯符号/数字不含日文，翻译层自会跳过。 */
    const 裸界面 = new Set(['点']);

    /* ---- 数据表的专有名词列 ----
     * referkey.php / gamegrouplist.php / mod_*_list.php 这类列表页，
     * 作品名、系列名、假名读音是裸 <td>，没有包在专有名词链接里。
     * 而「クリアシリーズ」（Clearシリーズ 的读音）这种整串都由词典词构成，
     * 靠内容无法与界面文字区分 —— 只能按所在列判断。 */
    const 专有列名 = /^(?:ID|名前|フリガナ|よみ|読み|グループ名|グループ名フリガナ|クリエイター名|ブランド名|ゲーム名|音楽名|曲名|サークル名|キャラクター名|ユーザー名|HPのタイトル|HPのURL)$/;

    /* 每个 table 只算一次受保护列的下标集合 */
    const 列缓存 = new WeakMap();
    function 受保护列(表) {
        let s = 列缓存.get(表);
        if (s) return s;
        s = new Set();
        /* 取第一行含 th 的作为表头 */
        const 行 = 表.rows;
        for (let i = 0; i < 行.length; i++) {
            const ths = 行[i].cells;
            let 有th = false;
            for (let j = 0; j < ths.length; j++) if (ths[j].tagName === 'TH') { 有th = true; break; }
            if (!有th) continue;
            for (let j = 0; j < ths.length; j++) {
                /* 表头可能已被翻译过，用 title 里保存的原文兜底 */
                const 文 = (ths[j].getAttribute('title') || ths[j].textContent || '').split(标记)[0].trim();
                if (专有列名.test(文)) s.add(j);
            }
            break;
        }
        列缓存.set(表, s);
        return s;
    }

    /* 注：判断「是否位于专有名词列」的逻辑已合并进 祖先状态()，
     * 在遇到 TD 时算一次并由子孙继承，避免逐文本节点跑 closest()。 */

    /* 这些 href 指向的链接文字「可能」是专有名词：作品名、品牌名、人名、用户名等。
     * 全站有 130+ 个端点，命名很杂（game.php / o_game.php / contents_game.php /
     * game_tag.php / user_favorites_creater.php …），按关键词匹配而不是逐个列举。
     *
     * 注意：href 只说明链接「指向哪」，不代表链接文字本身是专有名词。
     * 侧边菜单里的「サマリー」「ブランド別」也指向 user_*.php，却是纯界面文字。
     * 所以这个正则只是第一道筛子，还要配合「导航区放行」判断，见 界面区()。 */
    const 专有名词链接 = /[^\/]*(?:game|brand|creater|music|character|shokushu|gamegroup|group|user|pov|tag|memo|okazu|shiyou|kensaku|referkey)[^\/]*\.php/i;

    /* 导航/菜单区：这些容器里的链接文字一律是界面用语，不是专有名词。
     * 真实结构确认：#left_menu / #right_menu / #menu / #serch_menu。 */
    const 导航区 = '#left_menu,#right_menu,#menu,#serch_menu,#topmenu,.m-drawer,.m-topbar,nav';

    /* ---- 祖先状态缓存 ----
     * 性能关键：closest('#a,.b,#c,…') 这种长选择器每次都要沿祖先链重新匹配。
     * 大数据页（referkey.php 有 10 万个文本节点）逐节点调用会花掉 5~8 秒，把标签卡死。
     * 改为每个「元素」只判定一次，并从父元素继承结果：
     * 每元素一次廉价 matches()，整体从 O(节点数 × 深度 × 选择器) 降到 O(元素数)。 */
    let 祖先缓存 = new WeakMap();   // 会在大批量变更时整表重建，故用 let
    const 空状态 = { 禁: false, 混: false, 专有: false, 列: false, 导航: false };

    /* 受保护容器在全站都只出现少数几个（#memo、#sql、.comment_pagging …）。
     * 与其对 17 万个元素逐个跑长选择器 matches()（约 7µs/次，合计 1.2 秒以上），
     * 不如一次 querySelectorAll 找出这些根元素，之后只做 Set.has()（O(1)）。 */
    let 禁根 = new Set(), 混根 = new Set(), 导航根 = new Set();
    function 刷新根集() {
        try {
            禁根 = new Set(document.querySelectorAll(完全禁入));
            混根 = new Set(document.querySelectorAll(混合容器));
            导航根 = new Set(document.querySelectorAll(导航区));
        } catch (e) { /* 选择器异常时退化为空集，宁可少保护也不报错 */ }
    }

    function 祖先状态(el) {
        if (!el || el.nodeType !== 1) return 空状态;
        let s = 祖先缓存.get(el);
        if (s) return s;

        const 父 = 祖先状态(el.parentElement);

        /* 导航区标记向下继承：菜单里的链接文字都是界面用语 */
        const 导航 = 父.导航 || 导航根.has(el);

        let 专有 = 父.专有;
        /* 只在自身是 <a> 时读 href，不再沿链 closest('a')。
         * 导航区内的链接一律放行 —— 「サマリー」「ブランド別」也指向
         * user_*.php，但它们是菜单项，不是用户名。 */
        if (!专有 && !导航 && el.nodeName === 'A') {
            const href = el.getAttribute('href') || '';
            if (专有名词链接.test(href) && !界面化链接(el)) 专有 = true;
        }

        /* 专有名词列：只在遇到 TD 时算一次，结果由子孙继承。
         * 注意用 el.parentElement.parentElement 直接取，不要用 closest('table')：
         * closest 在 10 万个单元格上要花 1 秒以上（实测），而 TD 的祖先结构
         * 固定是 td → tr → (tbody|table)，顺着 parentElement 走是 O(1)。 */
        let 列 = 父.列;
        if (!列 && el.nodeName === 'TD') {
            const tr = el.parentElement;
            let 表 = tr && tr.parentElement;
            if (表 && 表.nodeName !== 'TABLE') 表 = 表.parentElement;   // 跳过 tbody/thead
            if (表 && 表.nodeName === 'TABLE') {
                const 集 = 受保护列(表);
                if (集.size && 集.has(el.cellIndex)) 列 = true;
            }
        }

        s = {
            禁: 父.禁 || 禁根.has(el),
            混: 父.混 || 混根.has(el),
            专有,
            列,
            导航,
        };
        祖先缓存.set(el, s);
        return s;
    }

    /**
     * 结构保护：判断某个文本节点是否应当保持原文。
     * @param node 文本节点
     * @param 父   它的父元素
     */
    function 受保护(node, 父) {
        if (!父 || 父.nodeType !== 1) return true;
        if (禁入标签.has(父.nodeName)) return true;

        const st = 祖先状态(父);
        if (st.禁) return true;      // 完全禁入容器
        if (st.专有) return true;    // 专有名词链接内

        /* 混合容器：只有直接躺在容器里的裸文本才是用户投稿。
         * 包在 <a>/<span class="play_time"> 等子元素里的是界面文字，放行。 */
        if (st.混) {
            /* 只有「直接躺在混合容器里」的裸文本才是用户投稿 */
            const 是裸文本 = 父.matches ? 父.matches(混合容器) : false;
            if (是裸文本 && !裸界面.has((node.nodeValue || '').trim())) return true;
        }

        /* 数据表的专有名词列（名前 / フリガナ / グループ名 …）：
         * 这些裸 <td> 里是作品名、系列名、假名读音。
         * 「クリアシリーズ」（Clearシリーズ 的读音）整串都由词典词构成，
         * 内容上与界面文字无法区分，只能按列位置保护。
         * 结果已在 祖先状态 里算好并被子孙继承。 */
        if (st.列) return true;

        return false;
    }

    /* 元素属性是否受保护（属性上不存在"裸文本"问题，按容器判断即可） */

        if (!el || el.nodeType !== 1) return true;
        if (禁入标签.has(el.nodeName)) return true;
        /* 同样走祖先状态缓存，避免逐元素跑长选择器 */
        const st = 祖先状态(el);
        if (st.禁) return true;
        if (el.nodeName === 'A') {
            const href = el.getAttribute('href') || '';
            if (专有名词链接.test(href) && !界面化链接(el)) return true;
        }
        return false;
    }

        /^次の\d+件/, /^前の\d+件/, /^対象/,
    ];
    function 界面化链接(a) {
        const t = (a.textContent || '').trim();
        return 界面化链接正则.some((re) => re.test(t));

         * 它们几乎能匹配任何含「の」「と」的串，若排在前面会抢掉

            const a = 译段(m[1]), b = 译段(m[2]);
            if (!a || !b) return null;
            /* B 是统计量时前置：得点の平均 → 平均得分 */

    /* =====================================================================
     * 3.5 碎片专用框架
     *    这些是站点「固定切法」产生的碎片，可以安全整体翻译。
     *    在 计算() 里先于碎片拦截运行 —— 否则会被一刀切掉，永远留日文。
     *    只放高置信度的固定模式，绝不放宽泛助词规则。
     * ===================================================================*/
    const 曲目类型 = {
        OP: '片头曲', ED: '片尾曲', '挿入歌': '插入曲', 'キャラソン': '角色歌',
        'イメージソング': '印象曲', 'テーマ': '主题曲', 'OPアレンジ': '片头曲改编',
        'グランドED': '真结局ED', 'EDアレンジ': '片尾曲改编', 'BGM': 'BGM',
    };
    const 碎片框架 = [
        /* 音乐页：「(作品名のOP)」被切成 "作品名" + "のOP)"，只译后半 */
        [/^の([^)）]+)\)$/, (m) => {
            const t = 曲目类型[m[1]];
            return t ? `的${t}）` : null;   // 只认已知曲目类型，避免误翻作品名
        }],
        /* 「キャラソン(」+ 角色名 + 「)」 */
        [/^([^(（]+)[(（]$/, (m) => {
            const t = 曲目类型[m[1]] || 词素[m[1]];
            return t ? `${t}（` : null;
        }],
        /* 表格里的独立词 */
        [/^する$/, () => '开启'],      // 设置页单选按钮，只在整串相等时生效
        [/^しない$/, () => '关闭'],
        [/^有り$/, () => '有'],
        [/^無し$/, () => '无'],
        [/^表紙へ$/, () => '返回封面'],
        [/^ゲームIDが$/, () => '游戏ID为'],
        [/^発売日が$/, () => '发售日为'],
        [/^(.+?)へ$/, (m) => { const t = 词素[m[1]]; return t ? `前往${t}` : null; }],

        /* ---- 外部站点 ID 列 ----
         * 结构是 <a>Getchu.com</a> + "のID"，站名在链接里、「のID」是裸文本。
         * 这类碎片以「の」开头会被拦截，必须在这里先接住。 */

            const 表 = { ID: 'ID', URL: 'URL', 'サブドメイン': '子域名', 'サブディレクトリ': '子目录', 'ページ': '页面', 'リンク': '链接' };
            return '的' + 表[m[1]];
        }],
        [/^([\w.\- ]+)の(ID|URL|サブドメイン|サブディレクトリ)$/, (m) => {
            const 表 = { ID: 'ID', URL: 'URL', 'サブドメイン': '子域名', 'サブディレクトリ': '子目录' };
            return `${m[1]} 的${表[m[2]]}`;   // 站名是拉丁文，原样保留
        }],

        /* ---- 以开引号结尾的前导碎片 ----
         * 结构：「似た得点をつけてるユーザーさんが「」+ <span>とても使えた</span> + 「」をつけた…」
         * 前导段以「が「」收尾，会被碎片拦截挡掉，导致这半句永远是日文。
         * 这里只接站点固定的几种前导句，翻完仍以「」结尾，与后段正常拼接。 */
        [/^似た得点をつけてるユーザー(?:さん)?が[「『]$/, () => '评分相近的用户打出「'],

        [/^さんのファン(?:\((\d+)人\))?$/, (m) => m[1] ? ` 的粉丝（${m[1]} 人）` : ' 的粉丝'],

        [/^[」』]するゲーム$/, () => '」的游戏'],
        [/^[」』]点をつけたゲーム一覧$/, () => '」分的游戏一览'],

        /* 「(职位)」这类括号内的制作职务 */
        [/^\(([^)]+)\)$/, (m) => {

        [/^へのたどり着き方$/, () => '的到达方式'],
        [/^←ココ$/, () => '←就是这里'],
        [/^(メディ倫|ソフ倫|映像倫)\(([\w]+)\)$/, (m) => `${译段(m[1]) || m[1]}（${m[2]}）`],

    /* AのB 中需要前置的统计量 */
    const 统计头 = new Set(['平均', '平均値', '中央値', '標準偏差', '総数', '合計', '推移', '変遷', '数', '件数']);

    /* =====================================================================
     * 4. 组合引擎
     *    词素按长度降序做最长匹配。核心原则：全成功才输出。
     *    只要句中出现一个不认识的日文片段，整条放弃翻译 —— 宁可留日文，
     *    也不吐出半中半日的夹生文本。
     * ===================================================================*/
    const 词素键 = Object.keys(词素).sort((a, b) => b.length - a.length);

    /* 日文假名或汉字 */
    const 有日文 = (t) => /[぀-ゟ゠-ヿ㐀-鿿]/.test(t);

    /* 允许原样穿过组合的 ASCII 词：站点里真实出现的技术缩写和平台名。
     * 不在此表里的拉丁文字一律当作专有名词（作品名/品牌名/系列名），
     * 遇到就整条放弃翻译。
     * 否则「IF MATEシリーズ」会被拼成「IF MATE系列」——
     * 拉丁主干被当成中性填充穿过，只有后缀被翻译，正是要避免的夹生输出。 */
    const ASCII白名单 = new Set([
        'POV', 'SQL', 'CG', 'BGM', 'OP', 'ED', 'CV', 'ID', 'DL', 'TV', 'PC', 'OS', 'UI',
        'DVD', 'BD', 'MAC', 'UMD', 'PG', 'RPG', 'VN', 'TBL', 'ETC', 'ADV', 'SLG', 'NVL',
        'NS', 'PS', 'PSP', 'PSV', 'FD', 'CD', 'SE', 'SD', 'MEME', 'ASMR', 'OVA', 'BL',
        'AI', 'NTR', 'H', 'X', 'W', 'D', 'G', 'MB', 'GB', 'KB', 'No', 'HP', 'URL',
        'Windows', 'Win', 'Win2k', 'XP', 'Vista', 'NT', 'FANZA', 'DMM', 'DLsite',
        'JCRC', 'CSA', 'Getchu', 'Steam', 'Switch', 'Android', 'iOS', 'ErogameScape',
    ]);

    /** 判断一段非日文片段能否原样穿过组合 */
    function 可穿过(片段) {
        const s = 片段.trim();
        if (!s) return true;
        /* 纯数字、符号、空白：安全 */
        if (!/[A-Za-z]/.test(s)) return true;
        /* 含拉丁字母：必须每个词都在白名单里（忽略数字与常见符号） */
        const 词 = s.split(/[^A-Za-z0-9]+/).filter(Boolean);
        return 词.every((w) => ASCII白名单.has(w) || ASCII白名单.has(w.replace(/\d+$/, '')) || /^\d+$/.test(w));
    }

    /**
     * 词素最长匹配组合。
     * @returns {string|null} 全部片段都能处理则返回译文，否则 null
     */
    function 组合(s) {
        let out = '';
        let i = 0;
        let 命中 = false;
        let 未知 = '';

        while (i < s.length) {
            let 匹配 = null;
            for (let k = 0; k < 词素键.length; k++) {
                const key = 词素键[k];
                if (s.startsWith(key, i)) { 匹配 = key; break; }
            }

            if (匹配) {
                /* 先结算之前攒下的未知片段 */
                if (未知) {
                    if (有日文(未知) || !可穿过(未知)) return 记未知(未知, s);
                    out += 未知;
                    未知 = '';
                }
                out += 词素[匹配];
                i += 匹配.length;
                命中 = true;
            } else {
                未知 += s[i];
                i++;
            }
        }
        if (未知) {
            if (有日文(未知) || !可穿过(未知)) return 记未知(未知, s);
            out += 未知;
        }
        return 命中 ? out : null;
    }

    function 记未知(片段, 全句) {
        if (设置.开发者模式) {
            const k = 片段.trim();
            if (k && 有日文(k)) {
                const rec = 未覆盖.get(k) || { 次数: 0, 例: 全句 };
                rec.次数++;
                未覆盖.set(k, rec);
            }
        }
        return null;
    }

    /* 翻译缓存：同一串在页面里往往出现几十次 */
    const 缓存 = new Map();

    /**
     * 翻译一段界面文本。
     * @returns {string|null} 无法可靠翻译时返回 null（调用方保持原文）
     */
    function 翻译(原) {
        if (原 == null) return null;
        const s = 原.trim();
        if (!s) return null;
        if (缓存.has(s)) return 缓存.get(s);

        let r = 计算(s);
        缓存.set(s, r);
        return r;
    }

    function 计算(s) {
        /* 没有日文就不用翻 */
        if (!有日文(s)) return null;
        /* 过长的基本是用户正文，交给结构层拦，这里再兜一层 */
        if (s.length > 60) return null;

        /* 站点固有名词保持原样 */
        if (固有名词.has(s)) return null;

        /* 4.0 碎片专用框架：先于碎片拦截运行。
         * 有些碎片是「站点固定切法」，能安全整体翻译（音乐页的「(◯◯のOP)」、
         * 表格里的「有り」「表紙へ」）。它们必须排在下面的拦截之前，
         * 否则会被一刀切掉，页面上永远留着日文。 */
        for (let i = 0; i < 碎片框架.length; i++) {
            const [re, fn] = 碎片框架[i];
            const m = s.match(re);
            if (!m) continue;
            let r;
            try { r = fn(m); } catch (e) { r = null; }
            if (r && r !== s) return r;
        }

        /* ---- 句子碎片一律不翻 ----
         * 页面里常把一句话用内联标签切开，例如：
         *   "似た得点をつけてるユーザーが「" + <span>必ず購入</span> + "」をつけたゲーム一覧"
         * 每段单独送进来时，助词收尾的碎片会被宽泛框架命中
         * （が → 为），拼出「…ユーザー为「必ず購入」を…」这种夹生结果。
         * 判定：以助词/开引号结尾，或以助词/闭引号开头 —— 说明这不是完整短语。 */
        /* 例外：もの / こと / ため 等形式名词本身以「の/と/め」结尾，
         * 是完整名词而非悬空助词（データ数が5以上になったもの）。 */
        if (/[はがをにでとやのへも]$/.test(s) && !/(?:もの|こと|ため|まで|ほど|など)$/.test(s)) return null;
        if (/[「『（(]$/.test(s)) return null;
        if (/^[」』）)]/.test(s)) return null;
        /* 首字是助词 → 碎片。但要排除以助词同形字开头的正常词：
         * とても（很）/ ともに / のみ（仅）/ には … 这些不是悬空助词。
         * 漏掉这个例外会把「とても使えた」这类整词也判成碎片。 */
        if (/^[はがをにでとやのへも]/.test(s) &&
            !/^(?:とても|ともに|とも|のみ|には|とは|での|もの|もし|もっと|もう|はじ|ので|なので|やや|より|わり)/.test(s)) return null;

        /* 五十音检索表头：ア / ア行 —— 假名本身是索引键，不能翻 */
        if (假名表头.test(s)) return s.endsWith('行') ? s.slice(0, -1) + '行' : null;

        /* 4.1 先套语法框架
         * 必须放在「整句」判断之前：「この文書は〜に生成されました。」这类
         * 固定句以「。」结尾，但确实要翻。 */
        for (let i = 0; i < 框架.length; i++) {
            const [re, fn] = 框架[i];
            const m = s.match(re);
            if (!m) continue;
            let r;
            try { r = fn(m); } catch (e) { r = null; }
            if (r && r !== s) return r;
        }

        /* 框架都没接住、又以句末标点结尾的长串，基本是说明文或用户正文，
         * 不做词素拼装（拼出来必然是夹生的）。 */
        if (/[。！？]$/.test(s) && s.length > 12) return null;

        /* 4.2 括号包裹：(同人) (非18禁) → 内部单独翻 */
        const 括号 = s.match(/^([（(\[［【「]+)(.+?)([）)\]］】」]+)$/);
        if (括号) {
            const inner = 翻译(括号[2]);
            if (inner) return '（' + inner + '）';
            return null;
        }

        /* 4.3 尾部标点剥离：発売日： → 発売日 + ： */
        const 尾 = s.match(/^(.+?)([：:]\s*)$/);
        if (尾) {
            const t = 翻译(尾[1]);
            if (t) return t + '：';
            return null;
        }

        /* 4.4 斜杠 / 中点分隔：POV/属性/タグ、音楽・歌 */
        if (/[\/／・]/.test(s)) {
            const 分隔 = s.split(/([\/／・])/);
            let ok = true, out = '';
            for (const 段 of 分隔) {
                if (/^[\/／・]$/.test(段)) { out += 段 === '・' ? '・' : '/'; continue; }
                /* 非日文段同样要过白名单：否则「Apocripha/0シリーズ」
                 * 会被切成 Apocripha + 0シリーズ，前半当中性穿过、后半被翻译，
                 * 拼出「Apocripha/0系列」这种夹生结果。 */
                if (!有日文(段)) { if (!可穿过(段)) { ok = false; break; } out += 段; continue; }
                const t = 翻译(段);
                if (!t) { ok = false; break; }
                out += t;
            }
            if (ok && out) return out;
        }

        /* 4.5 词素组合 */
        return 组合(s);
    }

    /* =====================================================================
     * 5. DOM 应用层
     * ===================================================================*/
    /* 标记 已在文件开头声明（受保护列 需要提前用到） */
    const 已处理 = new WeakSet();

    /** 翻译单个文本节点 */
    function 处理文本(node) {
        if (已处理.has(node)) return;
        已处理.add(node);

        const 原 = node.nodeValue;
        if (!原 || !原.trim() || !有日文(原)) return;

        const 父 = node.parentNode;
        if (受保护(node, 父)) return;

        const 译 = 翻译(原);
        if (!译) return;

        /* 保留原有前后空白，避免布局错位 */
        node.nodeValue = 原.match(/^\s*/)[0] + 译 + 原.match(/\s*$/)[0];

        if (设置.保留原文提示 && 父 && 父.nodeType === 1 && !父.hasAttribute('title')) {
            父.setAttribute('title', 原.trim());
        }
    }

    const 属性名 = ['title', 'alt', 'placeholder', 'aria-label', 'label'];

    /** 翻译元素属性 */
    function 处理属性(el) {
        if (!设置.翻译属性) return;
        if (属性受保护(el)) return;

        for (const a of 属性名) {
            if (!el.hasAttribute(a)) continue;
            const 原 = el.getAttribute(a);
            if (!原 || !有日文(原) || 原.includes(标记)) continue;
            const 译 = 翻译(原);
            if (!译) continue;
            el.setAttribute(a, a === 'title' && 设置.保留原文提示 ? 译 + 标记 + 原.trim() : 译);
        }

        /* 按钮类 input 的 value 是可见文字，输入框的不能动 */
        if (el.nodeName === 'INPUT') {
            const t = (el.type || '').toLowerCase();
            if (t === 'submit' || t === 'button' || t === 'reset') {
                const 原 = el.value;
                if (原 && 有日文(原)) {
                    const 译 = 翻译(原);
                    if (译) {
                        el.value = 译;
                        if (设置.保留原文提示 && !el.hasAttribute('title')) el.setAttribute('title', 原.trim());
                    }
                }
            }
        }
    }

    /** 建立带保护剪枝的 TreeWalker */
    function 建walker(根) {
        return document.createTreeWalker(根, NodeFilter.SHOW_ELEMENT | NodeFilter.SHOW_TEXT, {
            acceptNode(n) {
                if (n.nodeType === 1) {
                    if (禁入标签.has(n.nodeName)) return NodeFilter.FILTER_REJECT;
                    /* 整棵子树都受保护的，直接 REJECT 掉，连里面的文本节点都不访问。
                     * 这是大页面的性能关键：referkey.php 有 3.5 万行、10 万个
                     * 数据单元格，其中「名前/フリガナ」两列全是专有名词，
                     * 在这里剪掉可省下绝大部分遍历。
                     * 混合容器不能剪 —— 里面混着界面文字，仍需逐节点判断。 */
                    const st = 祖先状态(n);
                    if (st.禁 || st.列 || st.专有) return NodeFilter.FILTER_REJECT;
                    /* 元素本身只可能因「属性」需要翻译。没有任何属性的元素
                     * （表格里绝大多数 tr/td/span 都是）不可能有 title/alt/placeholder，
                     * 用 SKIP 跳过它自己但继续深入子节点 —— 省掉每元素 5 次
                     * hasAttribute + getAttribute，这是大表页的主要开销。 */
                    if (n.attributes.length === 0) return NodeFilter.FILTER_SKIP;
                    return NodeFilter.FILTER_ACCEPT;
                }
                /* 文本节点：不含日文的（空白、数字、纯 ASCII）直接不产出。
                 * 表格页里换行缩进产生的空白文本节点极多
                 * （referkey.php 有 17 万个），逐个进函数纯属浪费。 */
                const v = n.nodeValue;
                if (!v || !有日文(v)) return NodeFilter.FILTER_REJECT;
                return NodeFilter.FILTER_ACCEPT;
            },
        });
    }

    /* ---- 分片遍历 ----
     * 剖析结论：耗时几乎全在 DOM 遍历本身（翻译计算只占 0.1%），
     * 而站内确实存在 referkey.php（3.5 万行）这种超大表。
     * 与其继续压常数，不如按时间片让出主线程：
     * 每片最多跑 12ms，之后交还浏览器，页面就不会「卡住」。
     * 首屏可见区域先同步翻完，避免用户看到日文闪现。 */
    const 片时长 = 12;

    function 遍历(根, 分片) {
        if (!根) return;
        if (根.nodeType === 3) { 处理文本(根); return; }
        if (根.nodeType !== 1) return;
        if (禁入标签.has(根.nodeName)) return;

        处理属性(根);
        const walker = 建walker(根);

        /* 小页面（或明确要求同步）直接一次跑完，省掉调度开销 */
        if (!分片) {
            let n;
            while ((n = walker.nextNode())) {
                if (n.nodeType === 3) 处理文本(n); else 处理属性(n);
            }
            return;
        }

        /* 大页面：改用自己控制的显式栈 DFS，不用 TreeWalker。
         * 原因：TreeWalker 的内部遍历无法中断 —— 受保护子树被 REJECT 后
         * 不产出节点，挂在「产出数」上的时钟检查就永不触发，
         * referkey.php 会出现「0 次分片却占用 6 秒」。
         * 手写 DFS 能对每个「被访问」的节点计数，才能真正切片。 */
        const 栈 = [根];
        let 计 = 0;

        const 跑一片 = () => {
            const 现在 = () => (typeof performance !== 'undefined' ? performance.now() : Date.now());
            const 截止 = 现在() + 片时长;

            while (栈.length) {
                const n = 栈.pop();

                if (n.nodeType === 3) {
                    if (n.nodeValue && 有日文(n.nodeValue)) 处理文本(n);
                } else if (n.nodeType === 1) {
                    if (禁入标签.has(n.nodeName)) continue;
                    const st = 祖先状态(n);
                    if (st.禁 || st.列 || st.专有) continue;   // 整棵子树跳过
                    if (n.attributes.length) 处理属性(n);
                    /* 逆序入栈，保持文档顺序处理 */
                    const 子 = n.childNodes;
                    for (let i = 子.length - 1; i >= 0; i--) 栈.push(子[i]);
                }

                /* 每 512 个被访问节点查一次时钟 */
                if ((++计 & 0x1ff) === 0 && 现在() > 截止) { 让出(跑一片); return; }
            }
        };
        跑一片();
    }

    /* 优先用 requestIdleCallback，回落到 setTimeout(0) */
    function 让出(fn) {
        if (typeof requestIdleCallback === 'function') requestIdleCallback(() => fn(), { timeout: 200 });
        else setTimeout(fn, 0);
    }

    /* =====================================================================
     * 6. 启动与动态监听
     * ===================================================================*/
    let observer = null;
    let 队列 = [];
    let 定时 = null;

    function 冲刷() {
        定时 = null;
        const 批 = 队列; 队列 = [];
        if (!批.length) return;
        /* DOM 变化后受保护容器可能新增（AJAX 插入评论区等），
         * 根集必须重扫；祖先缓存以元素为键，新元素本来就没有缓存，
         * 但父链状态可能已过期，批量较大时整表重建更省心。 */
        刷新根集();
        if (批.length > 200) 祖先缓存 = new WeakMap();
        if (observer) observer.disconnect();
        try { for (const n of 批) 遍历(n); }
        catch (e) { console.error('[EGS] 翻译异常', e); }
        finally { if (observer) 监听(); }
    }

    function 入队(n) {
        if (!n) return;
        队列.push(n);
        if (定时 === null) 定时 = setTimeout(冲刷, 40);
    }

    const 监听配置 = { childList: true, subtree: true, characterData: true };
    function 监听() { if (observer) observer.observe(document.documentElement, 监听配置); }

    /* =====================================================================
     * 5.5 字体美化
     *    只改字体族与行高，不动布局尺寸 —— 站点是固定宽度表格布局，
     *    改字号/间距容易把统计表撑破。
     * ===================================================================*/
    function 注入样式() {
        if (!设置.美化字体) return;
        if (document.getElementById('egs-cn-font')) return;

        const css = `
/* ErogameScape 中文化：字体美化 */
body, table, td, th, div, span, p, li, a, h1, h2, h3, h4, h5,
input, select, button, textarea, caption, legend, label, option {
  font-family: ${字体栈} !important;
}
/* 正文行高略放宽，中文比日文更需要行距 */
body { line-height: 1.7 !important; -webkit-font-smoothing: antialiased; }
td, th, li { line-height: 1.65 !important; }
/* 长评/短评正文：读得更舒服 */
#memo, .break-word, .comment_pagging, #comment_main {
  line-height: 1.85 !important;
  letter-spacing: .01em;
}
/* 表格里的数字用等宽数字对齐，统计表更整齐 */
td, th { font-variant-numeric: tabular-nums; }
/* SQL / 代码区保持等宽，别被上面的规则覆盖 */
#sql, #sql_text, .sql_text, #usersql, pre, code, textarea#sql {
  font-family: "Cascadia Mono", "Consolas", "Sarasa Mono SC", "Microsoft YaHei Mono", monospace !important;
}
`;
        const st = document.createElement('style');
        st.id = 'egs-cn-font';
        st.textContent = css;
        (document.head || document.documentElement).appendChild(st);
    }

    function 启动() {
        if (!设置.启用) return;

        注入样式();
        const t0 = performance.now();
        刷新根集();              // 必须先扫出受保护容器根集，否则保护判定全部失效

        /* 大页面分片跑，避免冻结标签页；小页面同步跑完更快也无闪烁。
         * 阈值按元素数判断（referkey.php 有 17 万个元素）。 */
        const 元素数 = document.getElementsByTagName('*').length;
        遍历(document.documentElement, 元素数 > 8000);

        /* 页面标题：ErogameScape 的 title 是「页面名 ErogameScape -エロゲー批評空間-」 */
        if (document.title) {
            const 部分 = document.title.split(/\s+(?=ErogameScape)/);
            const 头 = 翻译(部分[0]);
            if (头) document.title = [头, ...部分.slice(1)].join(' ');
        }
        console.log(`[EGS中文化] 首屏 ${(performance.now() - t0).toFixed(1)}ms`);

        if (设置.监听动态) {
            observer = new MutationObserver((rs) => {
                for (const r of rs) {
                    if (r.type === 'characterData') { 已处理.delete(r.target); 入队(r.target); }
                    else for (const n of r.addedNodes) 入队(n);
                }
            });
            监听();
        }

        if (设置.开发者模式) 装开发工具();
    }

    /* =====================================================================
     * 7. 开发者工具：EGS.miss() 列出未覆盖的日文词素
     * ===================================================================*/
    function 装开发工具() {
        const api = {
            /** 列出本页所有引擎放弃翻译的日文，按出现频次降序 —— 直接拿来补词素表 */
            miss() {
                const arr = [...未覆盖.entries()].sort((a, b) => b[1].次数 - a[1].次数);
                console.log(`%c未覆盖词素 ${arr.length} 个`, 'color:#e60;font-weight:bold');
                console.table(arr.map(([k, v]) => ({ 片段: k, 次数: v.次数, 出现于: v.例 })));
                const obj = {};
                for (const [k] of arr) obj[k] = '';
                console.log(JSON.stringify(obj, null, 4));
                return obj;
            },
            /** 扫描整页，把没翻译成功的可见日文列出来（含结构保护外的漏网） */
            scan() {
                const 剩 = new Map();
                const w = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
                    acceptNode(n) {
                        const p = n.parentNode;
                        if (p && (禁入标签.has(p.nodeName) || (p.closest && p.closest(完全禁入)))) return NodeFilter.FILTER_REJECT;
                        return NodeFilter.FILTER_ACCEPT;
                    },
                });
                let n;
                while ((n = w.nextNode())) {
                    const s = (n.nodeValue || '').trim();
                    if (!s || !有日文(s) || s.length > 60) continue;
                    if (受保护(n, n.parentNode)) continue;
                    剩.set(s, (剩.get(s) || 0) + 1);
                }
                const arr = [...剩.entries()].sort((a, b) => b[1] - a[1]);
                console.log(`%c页面残留日文 ${arr.length} 条`, 'color:#e60;font-weight:bold');
                console.table(arr.map(([k, v]) => ({ 文本: k, 次数: v })));
                return arr;
            },
            /** 单条试译，看引擎判断 */
            t(s) { return { 输入: s, 输出: 翻译(s), 组合: 组合(s) }; },
            词素, 框架, 设置,
        };
        try { (typeof unsafeWindow !== 'undefined' ? unsafeWindow : window).EGS = api; }
        catch (e) { window.EGS = api; }
        console.log('[EGS中文化] 开发者模式：EGS.miss() / EGS.scan() / EGS.t("文本")');
    }

    /* =====================================================================
     * 8. 菜单与入口
     * ===================================================================*/
    if (typeof GM_registerMenuCommand === 'function') {
        for (const k of Object.keys(默认设置)) {
            GM_registerMenuCommand((设置[k] ? '☑ ' : '☐ ') + k, () => {
                设置[k] = !设置[k];
                写('EGS_cfg', 设置);
                location.reload();
            });
        }
    }

    if (document.readyState === 'loading') {
        /* document-start 注入：body 一出现就翻一次，压掉日文闪现 */
        const 抢 = setInterval(() => {
            /* 抢跑同样要先刷根集，否则这一遍的保护判定会全部失效 */
            /* 抢跑：样式要尽早注入，避免先按原字体渲染再跳变 */
            if (document.body) { clearInterval(抢); if (设置.启用) { 注入样式(); 刷新根集(); 遍历(document.body); } }
        }, 20);
        setTimeout(() => clearInterval(抢), 8000);
        document.addEventListener('DOMContentLoaded', 启动, { once: true });
    } else {
        启动();
    }

})();
